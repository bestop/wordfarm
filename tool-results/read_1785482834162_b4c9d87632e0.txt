    90→const shuffle = <T,>(arr: T[]): T[] => {
    91→  const a = [...arr];
    92→  for (let i = a.length - 1; i > 0; i--) {
    93→    const j = Math.floor(Math.random() * (i + 1));
    94→    [a[i], a[j]] = [a[j], a[i]];
    95→  }
    96→  return a;
    97→};

  1002→function trySpawnDrop(state: GameState, zombie: Zombie, cellW: number) {
  1003→  let dropChance = 0.2;
  1004→  if (zombie.type === 'cone') dropChance = 0.3;
  1005→  if (zombie.type === 'bucket') dropChance = 0.4;
  1006→  if (zombie.type === 'flag') dropChance = 0.55;
  1007→  if (Math.random() < dropChance) {
  1008→    const typeRand = Math.random();
  1009→    let type: Pickup['type'] = 'sun';
  1010→    let value = 25 + Math.floor(Math.random() * 20);
  1011→    if (zombie.type === 'bucket' || zombie.type === 'flag') value = 35 + Math.floor(Math.random() * 25);
  1012→    if (typeRand > 0.85) { type = 'double'; value = 0; }
  1013→    else if (typeRand > 0.7) { type = 'freeze'; value = 0; }
  1014→    state.pickups.push({
  1015→      id: uid(), x: zombie.x, row: zombie.row,
  1016→      type, timer: 8000, maxTimer: 8000, collected: false,
  1017→      bobPhase: Math.random() * Math.PI * 2, value,
  1018→    });
  1019→  }

  1022→function collectPickup(state: GameState, pk: Pickup, cellW: number, cellH: number, ox: number, oy: number) {
  1023→  pk.collected = true;
  1024→  const py = oy + pk.row * cellH + cellH / 2;
  1025→  if (pk.type === 'sun') {
  1026→    state.sun += pk.value;
  1027→    state.score += pk.value;
  1028→    state.floatingTexts.push({ id: uid(), x: pk.x, y: py, text: `+${pk.value}☀️`, color: '#FFD54F', timer: 1000, maxTimer: 1000 });
  1029→  } else if (pk.type === 'double') {
  1030→    state.doubleDamageEnd = Date.now() + 6000;
  1031→    state.floatingTexts.push({ id: uid(), x: pk.x, y: py, text: '⚡双倍伤害 6s!', color: '#FF6F00', timer: 1500, maxTimer: 1500 });
  1032→  } else if (pk.type === 'freeze') {
  1033→    state.zombies.forEach(z => { if (!z.dead) { z.slowed = true; z.slowTimer = 4000; } });
  1034→    state.floatingTexts.push({ id: uid(), x: pk.x, y: py, text: '🧊全屏冰冻 4s!', color: '#29B6F6', timer: 1500, maxTimer: 1500 });
  1035→  }
  1036→  spawnParticles(state, pk.x, py, pk.type === 'sun' ? '#FFD54F' : pk.type === 'double' ? '#FF6F00' : '#29B6F6', 15);
  1037→}
