    92→const shuffle = <T,>(arr: T[]): T[] => {
    93→  const a = [...arr];
    94→  for (let i = a.length - 1; i > 0; i--) {
    95→    const j = Math.floor(Math.random() * (i + 1));
    96→    [a[i], a[j]] = [a[j], a[i]];
    97→  }
    98→  return a;
    99→};
