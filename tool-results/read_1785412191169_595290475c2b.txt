   283→  // ---- Game loop ----
   284→  const gameLoop = useCallback(() => {

   288→    if (state.phase !== 'playing') {
   289→      // Still render one last frame

   302→    // -- Spawn wave zombies --

   321→          break;
   322→        }

   348→    for (const zombie of state.zombies) {

   371→          }
   372→        }
   373→      } else {

   378→        gameOver = true;
   379→      }

   431→      if (!proj.active) continue;

   435→      );

   437→        proj.active = false;
   438→        hitZombie.hp -= proj.damage;
   439→        if (proj.slow) { hitZombie.slowed = true; hitZombie.slowTimer = 3000; }

   441→          hitZombie.dead = true; hitZombie.deathTimer = 500;
   442→          state.totalKills++; state.score += 50;
   443→        }
   444→        state.floatingTexts.push({
   445→          id: uid(), x: proj.x, y: oy + proj.row * cellH + cellH * 0.25,
   446→          text: `-${proj.damage}`, color: proj.slow ? '#00E5FF' : '#FF6600',
   447→          timer: 600, maxTimer: 600,
   448→        });
   449→      }
   450→      if (proj.x > ox + (GRID_COLS + 1) * cellW) proj.active = false;
   451→    }

   453→

   455→    for (const sun of state.suns) {
   456→      if (sun.collected) {
   457→        sun.opacity -= dt / 300;
   458→        sun.scale += dt / 200;
   459→        continue;
   460→      }
   461→      if (sun.y < sun.targetY) sun.y = Math.min(sun.y + dt * 0.05, sun.targetY);
   462→      if (sun.scale < 1) sun.scale = Math.min(1, sun.scale + dt / 300);
   463→      sun.timer -= dt;
   464→    }
   465→    state.suns = state.suns.filter(s => s.timer > 0 && s.opacity > 0);
   466→

   468→    if (now >= state.nextNaturalSun) {
   469→      state.nextNaturalSun = now + NATURAL_SUN_INTERVAL;

   471→      state.suns.push({
   472→        id: uid(), x: sx, y: -20,
   473→        targetY: oy + Math.random() * (GRID_ROWS * cellH),
   474→        value: NATURAL_SUN_VALUE, timer: 10000,
   475→        collected: false, opacity: 1, scale: 0,
   476→      });
   477→    }

   485→

   488→      ft.y -= dt * 0.04;

   490→    state.floatingTexts = state.floatingTexts.filter(ft => ft.timer > 0);
   491→

   493→    if (state.currentQuiz) {

   503→          state.comboCount = 0;
