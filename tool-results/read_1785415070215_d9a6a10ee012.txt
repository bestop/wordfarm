     1→'use client';
     2→
     3→import React, { useCallback, useEffect, useRef, useState } from 'react';
     4→import {
     5→  WORD_BANK, PLANT_DEFS, PLANT_ORDER, ZOMBIE_DEFS,
     6→  GRID_COLS, GRID_ROWS, STARTING_SUN, QUIZ_SUN_REWARD,
     7→  QUIZ_TIME_LIMIT, QUIZ_COOLDOWN, NATURAL_SUN_INTERVAL,
     8→  NATURAL_SUN_VALUE, ZOMBIE_SPEED_BOOST, ZOMBIE_SPEED_BOOST_DURATION,
     9→  WAVE_CONFIGS,
    10→} from '@/game/data';
    11→import type { PlantType, ZombieType, Word } from '@/game/data';
    12→
    13→// ============ Internal Types ============
    14→type GamePhase = 'menu' | 'playing' | 'gameover' | 'victory';
    15→
    16→interface Plant {
    17→  id: string; type: PlantType; row: number; col: number;
    18→  hp: number; maxHp: number; lastAttack: number; lastSun: number;
    19→  animPhase: number;
    20→}
    21→
    22→interface Zombie {
    23→  id: string; type: ZombieType; row: number; x: number;
    24→  hp: number; maxHp: number; speed: number; baseSpeed: number;
    25→  eating: boolean; slowed: boolean; slowTimer: number;
    26→  lastHit: number; animPhase: number; dead: boolean; deathTimer: number;
    27→}
    28→
    29→interface Projectile {
    30→  id: string; row: number; x: number; speed: number;
    31→  damage: number; slow: boolean; active: boolean;
    32→}
    33→
    34→interface Sun {
    35→  id: string; x: number; y: number; targetY: number;
    36→  value: number; timer: number; collected: boolean;
    37→  opacity: number; scale: number;
    38→}
    39→
    40→interface Explosion {
    41→  id: string; x: number; y: number; radius: number;
    42→  maxRadius: number; timer: number; maxTimer: number;
    43→}
    44→
    45→interface FloatingText {
    46→  id: string; x: number; y: number; text: string;
    47→  color: string; timer: number; maxTimer: number;
    48→}
    49→
    50→interface WordQuestion {
    51→  word: Word; options: string[]; correctIndex: number;
    52→  timer: number; answered: boolean; wasCorrect?: boolean;
    53→}
    54→
    55→interface GameState {
    56→  phase: GamePhase;
    57→  sun: number; score: number; wave: number;
    58→  plants: Plant[]; zombies: Zombie[]; projectiles: Projectile[];
    59→  suns: Sun[]; explosions: Explosion[]; floatingTexts: FloatingText[];
    60→  selectedPlant: PlantType | null;
    61→  currentQuiz: WordQuestion | null;
    62→  quizCooldown: number;
    63→  wordsAnswered: number; wordsCorrect: number;
    64→  waveStartTime: number; waveZombiesSpawned: number;
    65→  totalKills: number; lastTime: number;
    66→  nextNaturalSun: number;
    67→  zombieSpeedBoostEnd: number;
    68→  usedWordIndices: Set<number>;
    69→  shakeTimer: number;
    70→  comboCount: number;
    71→}
    72→
    73→// ============ Helpers ============
    74→let _idCounter = 0;
    75→const uid = () => `e${++_idCounter}`;
    76→
    77→const shuffle = <T,>(arr: T[]): T[] => {
    78→  const a = [...arr];
    79→  for (let i = a.length - 1; i > 0; i--) {
    80→    const j = Math.floor(Math.random() * (i + 1));
    81→    [a[i], a[j]] = [a[j], a[i]];
    82→  }
    83→  return a;
    84→};
    85→
    86→// ============ Component ============
    87→export default function PvZGame() {
    88→  const canvasRef = useRef<HTMLCanvasElement>(null);
    89→  const containerRef = useRef<HTMLDivElement>(null);
    90→  const gs = useRef<GameState | null>(null);
    91→  const rafRef = useRef<number>(0);
    92→  const [phase, setPhase] = useState<GamePhase>('menu');
    93→  const [uiTick, setUiTick] = useState(0);
    94→  const forceUpdate = useCallback(() => setUiTick(t => t + 1), []);
    95→
    96→  // Canvas dimensions
    97→  const dims = useRef({ w: 900, h: 500, cellW: 80, cellH: 90, ox: 30, oy: 15 });
    98→
    99→  const recalcDims = useCallback(() => {
   100→    const canvas = canvasRef.current;
   101→    const container = containerRef.current;
   102→    if (!canvas || !container) return;
   103→    const rect = container.getBoundingClientRect();
   104→    const dpr = window.devicePixelRatio || 1;
   105→    const w = rect.width;
   106→    const h = rect.height;
   107→    canvas.width = w * dpr;
   108→    canvas.height = h * dpr;
   109→    canvas.style.width = `${w}px`;
   110→    canvas.style.height = `${h}px`;
   111→    const ctx = canvas.getContext('2d');
   112→    if (ctx) ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   113→    const cellW = (w * 0.9) / GRID_COLS;
   114→    const cellH = (h * 0.95) / GRID_ROWS;
   115→    const ox = w * 0.07;
   116→    const oy = h * 0.025;
   117→    dims.current = { w, h, cellW, cellH, ox, oy };
   118→  }, []);
   119→
   120→  // ---- Init game state ----
   121→  const initGame = useCallback(() => {
   122→    _idCounter = 0;
   123→    const state: GameState = {
   124→      phase: 'playing', sun: STARTING_SUN, score: 0, wave: 0,
   125→      plants: [], zombies: [], projectiles: [], suns: [],
   126→      explosions: [], floatingTexts: [],
   127→      selectedPlant: null, currentQuiz: null, quizCooldown: 0,
   128→      wordsAnswered: 0, wordsCorrect: 0,
   129→      waveStartTime: Date.now(), waveZombiesSpawned: 0,
   130→      totalKills: 0, lastTime: Date.now(),
   131→      nextNaturalSun: Date.now() + 8000,
   132→      zombieSpeedBoostEnd: 0, usedWordIndices: new Set(),
   133→      shakeTimer: 0, comboCount: 0,
   134→    };
   135→    gs.current = state;
   136→    setPhase('playing');
   137→    generateQuiz(state);
   138→    forceUpdate();
   139→  }, [forceUpdate]);
   140→
   141→  // ---- Generate quiz ----
   142→  const generateQuiz = useCallback((state: GameState) => {
   143→    // First 3 questions are always easy difficulty
   144→    const difficultyFilter = state.wordsAnswered < 3 ? 1 : 0;
   145→    const available = WORD_BANK.map((_, i) => i).filter(
   146→      i => !state.usedWordIndices.has(i) && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
   147→    );
   148→    if (available.length < 4) state.usedWordIndices.clear();
   149→    const pool = available.length >= 4
   150→      ? available
   151→      : WORD_BANK.map((_, i) => i).filter(i => !difficultyFilter || WORD_BANK[i].difficulty === 1);
   152→    const shuffled = shuffle(pool);
   153→    const correctIdx = shuffled[0];
   154→    const correctWord = WORD_BANK[correctIdx];
   155→    state.usedWordIndices.add(correctIdx);
   156→    const wrongPool = WORD_BANK.map((_, i) => i).filter(
   157→      i => i !== correctIdx && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
   158→    );
   159→    const wrongShuffled = shuffle(wrongPool).slice(0, 3);
   160→    const options = shuffle([correctWord.zh, ...wrongShuffled.map(i => WORD_BANK[i].zh)]);
   161→    state.currentQuiz = {
   162→      word: correctWord, options,
   163→      correctIndex: options.indexOf(correctWord.zh),
   164→      timer: QUIZ_TIME_LIMIT, answered: false, wasCorrect: undefined,
   165→    };
   166→    forceUpdate();
   167→  }, [forceUpdate]);
   168→
   169→  // ---- Handle quiz answer ----
   170→  const handleAnswer = useCallback((index: number) => {
   171→    const state = gs.current;
   172→    if (!state || !state.currentQuiz || state.currentQuiz.answered || state.phase !== 'playing') return;
   173→    const quiz = state.currentQuiz;
   174→    quiz.answered = true;
   175→    const isCorrect = index === quiz.correctIndex;
   176→    quiz.wasCorrect = isCorrect;
   177→    state.wordsAnswered++;
   178→    const now = Date.now();
   179→    const { w, h } = dims.current;
   180→
   181→    if (isCorrect) {
   182→      state.wordsCorrect++;
   183→      state.comboCount++;
   184→      const reward = QUIZ_SUN_REWARD[quiz.word.difficulty];
   185→      const comboBonus = state.comboCount >= 3 ? Math.floor(reward * 0.5) : 0;
   186→      state.sun += reward + comboBonus;
   187→      state.score += reward + comboBonus;
   188→      state.floatingTexts.push({
   189→        id: uid(), x: w / 2, y: h - 80,
   190→        text: `+${reward + comboBonus} ☀️${comboBonus > 0 ? ' 连击x' + state.comboCount : ''}`,
   191→        color: '#FFD700', timer: 1500, maxTimer: 1500,
   192→      });
   193→    } else {
   194→      state.comboCount = 0;
   195→      state.zombieSpeedBoostEnd = now + ZOMBIE_SPEED_BOOST_DURATION;
   196→      state.shakeTimer = 300;
   197→      state.floatingTexts.push({
   198→        id: uid(), x: w / 2, y: h - 80,
   199→        text: '答错了! 僵尸加速!', color: '#FF4444', timer: 1500, maxTimer: 1500,
   200→      });
   201→    }
   202→    state.quizCooldown = QUIZ_COOLDOWN;
   203→    forceUpdate();
   204→  }, [forceUpdate]);
   205→
   206→  // ---- Canvas click (plant placement) ----
   207→  const handleCanvasClick = useCallback((e: React.MouseEvent | React.TouchEvent) => {
   208→    const state = gs.current;
   209→    if (!state || state.phase !== 'playing' || !state.selectedPlant) return;
   210→    const canvas = canvasRef.current;
   211→    if (!canvas) return;
   212→    const rect = canvas.getBoundingClientRect();
   213→    let clientX: number, clientY: number;
   214→    if ('touches' in e) {
   215→      if (e.touches.length === 0) return;
   216→      clientX = e.touches[0].clientX;
   217→      clientY = e.touches[0].clientY;
   218→    } else {
   219→      clientX = e.clientX;
   220→      clientY = e.clientY;
   221→    }
   222→    const x = clientX - rect.left;
   223→    const y = clientY - rect.top;
   224→    const { cellW, cellH, ox, oy } = dims.current;
   225→    const col = Math.floor((x - ox) / cellW);
   226→    const row = Math.floor((y - oy) / cellH);
   227→    if (col < 0 || col >= GRID_COLS || row < 0 || row >= GRID_ROWS) return;
   228→    if (state.plants.some(p => p.row === row && p.col === col)) return;
   229→    const def = PLANT_DEFS[state.selectedPlant];
   230→    if (state.sun < def.cost) {
   231→      state.floatingTexts.push({
   232→        id: uid(), x, y, text: '阳光不足!', color: '#FF4444', timer: 1000, maxTimer: 1000,
   233→      });
   234→      return;
   235→    }
   236→    state.sun -= def.cost;
   237→    const plant: Plant = {
   238→      id: uid(), type: state.selectedPlant, row, col,
   239→      hp: def.hp, maxHp: def.hp,
   240→      lastAttack: Date.now(), lastSun: Date.now(), animPhase: 0,
   241→    };
   242→    state.plants.push(plant);
   243→
   244→    // Cherry bomb: delayed explosion
   245→    if (def.explosive) {
   246→      const capturedState = state;
   247→      const capturedPlant = plant;
   248→      const capturedCol = col;
   249→      const capturedRow = row;
   250→      setTimeout(() => {
   251→        if (capturedState.phase !== 'playing') return;
   252→        const { ox: oX, oy: oY, cellW: cW, cellH: cH } = dims.current;
   253→        const cx = oX + capturedCol * cW + cW / 2;
   254→        const cy = oY + capturedRow * cH + cH / 2;
   255→        capturedState.explosions.push({
   256→          id: uid(), x: cx, y: cy, radius: 0,
   257→          maxRadius: cW * 2, timer: 0, maxTimer: 600,
   258→        });
   259→        capturedState.zombies.forEach(z => {
   260→          if (!z.dead && Math.abs(z.row - capturedRow) <= 1) {
   261→            const zy = oY + z.row * cH + cH / 2;
   262→            const dist = Math.sqrt((z.x - cx) ** 2 + (zy - cy) ** 2);
   263→            if (dist < cW * 2.5) {
   264→              z.hp -= 1800;
   265→              if (z.hp <= 0) {
   266→                z.dead = true; z.deathTimer = 500;
   267→                capturedState.totalKills++; capturedState.score += 50;
   268→              }
   269→            }
   270→          }
   271→        });
   272→        capturedState.plants = capturedState.plants.filter(p => p.id !== capturedPlant.id);
   273→        forceUpdate();
   274→      }, 500);
   275→    }
   276→    forceUpdate();
   277→  }, [forceUpdate]);
   278→
   279→  // ---- Sun click ----
   280→  const handleSunClick = useCallback((sunId: string, e: React.MouseEvent) => {
   281→    e.stopPropagation();
   282→    const state = gs.current;
   283→    if (!state) return;
   284→    const sun = state.suns.find(s => s.id === sunId);
   285→    if (!sun || sun.collected) return;
   286→    sun.collected = true;
   287→    state.sun += sun.value;
   288→    forceUpdate();
   289→  }, [forceUpdate]);
   290→
   291→  // ---- Game loop ----
   292→  const gameLoop = useCallback(() => {
   293→    const state = gs.current;
   294→    const canvas = canvasRef.current;
   295→    if (!state || !canvas) return;
   296→    if (state.phase !== 'playing') {
   297→      // Still render one last frame
   298→      return;
   299→    }
   300→    const ctx = canvas.getContext('2d');
   301→    if (!ctx) return;
   302→
   303→    const now = Date.now();
   304→    const dt = Math.min(now - state.lastTime, 50);
   305→    state.lastTime = now;
   306→    const { w, h, cellW, cellH, ox, oy } = dims.current;
   307→    const isBoosted = now < state.zombieSpeedBoostEnd;
   308→    const speedMult = isBoosted ? ZOMBIE_SPEED_BOOST : 1;
   309→
   310→    // -- Spawn wave zombies --
   311→    const waveConfig = WAVE_CONFIGS[state.wave];
   312→    if (waveConfig) {
   313→      const waveElapsed = now - state.waveStartTime;
   314→      for (let i = state.waveZombiesSpawned; i < waveConfig.zombies.length; i++) {
   315→        if (waveElapsed >= waveConfig.zombies[i].delay) {
   316→          state.waveZombiesSpawned = i + 1;
   317→          const zc = waveConfig.zombies[i];
   318→          const def = ZOMBIE_DEFS[zc.type];
   319→          const row = zc.row ?? Math.floor(Math.random() * GRID_ROWS);
   320→          state.zombies.push({
   321→            id: uid(), type: zc.type, row,
   322→            x: ox + (GRID_COLS + 0.8) * cellW,
   323→            hp: def.hp, maxHp: def.hp, speed: def.speed, baseSpeed: def.speed,
   324→            eating: false, slowed: false, slowTimer: 0,
   325→            lastHit: now, animPhase: Math.random() * Math.PI * 2,
   326→            dead: false, deathTimer: 0,
   327→          });
   328→        } else {
   329→          break;
   330→        }
   331→      }
   332→    }
   333→
   334→    // -- Check wave complete --
   335→    const allSpawned = waveConfig ? state.waveZombiesSpawned >= waveConfig.zombies.length : true;
   336→    const allDead = state.zombies.length === 0 || state.zombies.every(z => z.dead);
   337→    if (allSpawned && allDead && state.zombies.length > 0) {
   338→      if (state.wave < WAVE_CONFIGS.length - 1) {
   339→        state.wave++;
   340→        state.waveStartTime = now;
   341→        state.waveZombiesSpawned = 0;
   342→        state.floatingTexts.push({
   343→          id: uid(), x: w / 2, y: h / 2,
   344→          text: `第 ${state.wave + 1} 波!`, color: '#FF6600', timer: 2000, maxTimer: 2000,
   345→        });
   346→        forceUpdate();
   347→      } else {
   348→        state.phase = 'victory';
   349→        setPhase('victory');
   350→        return;
   351→      }
   352→    }
   353→
   354→    // -- Update zombies --
   355→    let gameOver = false;
   356→    for (const zombie of state.zombies) {
   357→      if (zombie.dead) {
   358→        zombie.deathTimer -= dt;
   359→        continue;
   360→      }
   361→      if (zombie.slowed) {
   362→        zombie.slowTimer -= dt;
   363→        if (zombie.slowTimer <= 0) zombie.slowed = false;
   364→      }
   365→      zombie.animPhase += dt * 0.005 * (zombie.slowed ? 0.5 : 1);
   366→      const currentSpeed = zombie.baseSpeed * speedMult * (zombie.slowed ? 0.5 : 1);
   367→      const eatingPlant = state.plants.find(p => {
   368→        if (p.row !== zombie.row) return false;
   369→        const plantX = ox + p.col * cellW + cellW / 2;
   370→        return Math.abs(zombie.x - plantX) < cellW * 0.35;
   371→      });
   372→      if (eatingPlant) {
   373→        zombie.eating = true;
   374→        if (now - zombie.lastHit > 1000) {
   375→          zombie.lastHit = now;
   376→          eatingPlant.hp -= 100;
   377→          if (eatingPlant.hp <= 0) {
   378→            state.plants = state.plants.filter(p => p.id !== eatingPlant.id);
   379→          }
   380→        }
   381→      } else {
   382→        zombie.eating = false;
   383→        zombie.x -= currentSpeed * (dt / 1000);
   384→      }
   385→      if (zombie.x < ox - cellW * 0.5) {
   386→        gameOver = true;
   387→      }
   388→    }
   389→    state.zombies = state.zombies.filter(z => !(z.dead && z.deathTimer <= 0));
   390→
   391→    if (gameOver) {
   392→      state.phase = 'gameover';
   393→      setPhase('gameover');
   394→      return;
   395→    }
   396→
   397→    // -- Update plants --
   398→    for (const plant of state.plants) {
   399→      plant.animPhase += dt * 0.003;
   400→      const def = PLANT_DEFS[plant.type];
   401→      if (def.sunProduction && def.sunInterval && now - plant.lastSun >= def.sunInterval) {
   402→        plant.lastSun = now;
   403→        const px = ox + plant.col * cellW + cellW / 2;
   404→        const py = oy + plant.row * cellH + cellH / 2;
   405→        state.suns.push({
   406→          id: uid(), x: px, y: py - 10,
   407→          targetY: py + cellH * 0.3 + Math.random() * 20,
   408→          value: def.sunProduction, timer: 8000,
   409→          collected: false, opacity: 1, scale: 0,
   410→        });
   411→      }
   412→      if (def.attack && def.attackSpeed) {
   413→        const hasZombie = state.zombies.some(
   414→          z => !z.dead && z.row === plant.row && z.x > ox + plant.col * cellW
   415→        );
   416→        if (hasZombie && now - plant.lastAttack >= def.attackSpeed) {
   417→          plant.lastAttack = now;
   418→          const px = ox + plant.col * cellW + cellW * 0.7;
   419→          state.projectiles.push({
   420→            id: uid(), row: plant.row, x: px, speed: 250,
   421→            damage: def.attack, slow: !!def.slowEffect, active: true,
   422→          });
   423→          if (def.doubleShot) {
   424→            setTimeout(() => {
   425→              if (gs.current?.phase === 'playing') {
   426→                gs.current.projectiles.push({
   427→                  id: uid(), row: plant.row, x: px, speed: 250,
   428→                  damage: def.attack, slow: false, active: true,
   429→                });
   430→              }
   431→            }, 150);
   432→          }
   433→        }
   434→      }
   435→    }
   436→
   437→    // -- Update projectiles --
   438→    for (const proj of state.projectiles) {
   439→      if (!proj.active) continue;
   440→      proj.x += proj.speed * (dt / 1000);
   441→      const hitZombie = state.zombies.find(
   442→        z => !z.dead && z.row === proj.row && Math.abs(z.x - proj.x) < cellW * 0.25
   443→      );
   444→      if (hitZombie) {
   445→        proj.active = false;
   446→        hitZombie.hp -= proj.damage;
   447→        if (proj.slow) { hitZombie.slowed = true; hitZombie.slowTimer = 3000; }
   448→        if (hitZombie.hp <= 0) {
   449→          hitZombie.dead = true; hitZombie.deathTimer = 500;
   450→          state.totalKills++; state.score += 50;
   451→        }
   452→        state.floatingTexts.push({
   453→          id: uid(), x: proj.x, y: oy + proj.row * cellH + cellH * 0.25,
   454→          text: `-${proj.damage}`, color: proj.slow ? '#00E5FF' : '#FF6600',
   455→          timer: 600, maxTimer: 600,
   456→        });
   457→      }
   458→      if (proj.x > ox + (GRID_COLS + 1) * cellW) proj.active = false;
   459→    }
   460→    state.projectiles = state.projectiles.filter(p => p.active);
   461→
   462→    // -- Update suns --
   463→    for (const sun of state.suns) {
   464→      if (sun.collected) {
   465→        sun.opacity -= dt / 300;
   466→        sun.scale += dt / 200;
   467→        continue;
   468→      }
   469→      if (sun.y < sun.targetY) sun.y = Math.min(sun.y + dt * 0.05, sun.targetY);
   470→      if (sun.scale < 1) sun.scale = Math.min(1, sun.scale + dt / 300);
   471→      sun.timer -= dt;
   472→    }
   473→    state.suns = state.suns.filter(s => s.timer > 0 && s.opacity > 0);
   474→
   475→    // -- Natural sun drops --
   476→    if (now >= state.nextNaturalSun) {
   477→      state.nextNaturalSun = now + NATURAL_SUN_INTERVAL;
   478→      const sx = ox + Math.random() * (GRID_COLS * cellW);
   479→      state.suns.push({
   480→        id: uid(), x: sx, y: -20,
   481→        targetY: oy + Math.random() * (GRID_ROWS * cellH),
   482→        value: NATURAL_SUN_VALUE, timer: 10000,
   483→        collected: false, opacity: 1, scale: 0,
   484→      });
   485→    }
   486→
   487→    // -- Update effects --
   488→    for (const exp of state.explosions) {
   489→      exp.timer += dt;
   490→      exp.radius = (exp.timer / exp.maxTimer) * exp.maxRadius;
   491→    }
   492→    state.explosions = state.explosions.filter(e => e.timer < e.maxTimer);
   493→
   494→    for (const ft of state.floatingTexts) {
   495→      ft.timer -= dt;
   496→      ft.y -= dt * 0.04;
   497→    }
   498→    state.floatingTexts = state.floatingTexts.filter(ft => ft.timer > 0);
   499→
   500→    // -- Update quiz --
   501→    if (state.currentQuiz) {
   502→      if (state.currentQuiz.answered) {
   503→        state.quizCooldown -= dt;
   504→        if (state.quizCooldown <= 0) generateQuiz(state);
   505→      } else {
