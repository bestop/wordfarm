   954→export default function PvZGame() {
   955→  const canvasRef = useRef<HTMLCanvasElement>(null);
   956→  const containerRef = useRef<HTMLDivElement>(null);
   957→  const gs = useRef<GameState | null>(null);

   962→  const dims = useRef({ w: 900, h: 500, cellW: 80, cellH: 90, ox: 30, oy: 15 });
   963→
   964→  const recalcDims = useCallback(() => {
   965→    const canvas = canvasRef.current;

   967→    if (!canvas || !container) return;
   968→    const rect = container.getBoundingClientRect();
   969→    const dpr = window.devicePixelRatio || 1;
   970→    const w = rect.width; const h = rect.height;
   971→    canvas.width = w * dpr; canvas.height = h * dpr;

   973→    const ctx = canvas.getContext('2d');

   976→    const cellH = (h * 0.95) / GRID_ROWS;
   977→    const ox = w * 0.07; const oy = h * 0.025;
   978→    dims.current = { w, h, cellW, cellH, ox, oy };
   979→  }, []);
   980→
   981→  const initGame = useCallback(() => {
   982→    _idCounter = 0;
   983→    const state: GameState = {
   984→      phase: 'playing', sun: STARTING_SUN, score: 0, wave: 0,
   985→      plants: [], zombies: [], projectiles: [],
   986→      explosions: [], floatingTexts: [],
   987→      selectedPlant: null, currentQuiz: null, quizCooldown: 0,
   988→      wordsAnswered: 0, wordsCorrect: 0,
   989→      waveStartTime: Date.now(), waveZombiesSpawned: 0,
   990→      totalKills: 0, lastTime: Date.now(),
   991→      zombieSpeedBoostEnd: 0, usedWordIndices: new Set(),
   992→      shakeTimer: 0, comboCount: 0,
   993→    };
   994→    gs.current = state;
   995→    setPhase('playing');
   996→    generateQuiz(state);
   997→    forceUpdate();
   998→  }, [forceUpdate]);
   999→
  1000→  const generateQuiz = useCallback((state: GameState) => {
  1001→    const difficultyFilter = state.wordsAnswered < 3 ? 1 : 0;
  1002→    let available = WORD_BANK.map((_, i) => i).filter(
  1003→      i => !state.usedWordIndices.has(i) && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
  1004→    );
  1005→    if (available.length < 4) { state.usedWordIndices.clear(); }
  1006→    available = WORD_BANK.map((_, i) => i).filter(
  1007→      i => !state.usedWordIndices.has(i) && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
  1008→    );
  1009→    if (available.length < 4) available = WORD_BANK.map((_, i) => i);
  1010→    const shuffled = shuffle(available);
  1011→    const correctIdx = shuffled[0];
  1012→    const correctWord = WORD_BANK[correctIdx];
  1013→    state.usedWordIndices.add(correctIdx);
  1014→    const wrongPool = WORD_BANK.map((_, i) => i).filter(
  1015→      i => i !== correctIdx && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
  1016→    );
  1017→    const wrongShuffled = shuffle(wrongPool).slice(0, 3);
  1018→    const options = shuffle([correctWord.zh, ...wrongShuffled.map(i => WORD_BANK[i].zh)]);
  1019→    state.currentQuiz = {
  1020→      word: correctWord, options,
  1021→      correctIndex: options.indexOf(correctWord.zh),
  1022→      timer: QUIZ_TIME_LIMIT, answered: false, wasCorrect: undefined,
  1023→    };
  1024→    forceUpdate();
  1025→  }, [forceUpdate]);
  1026→
  1027→  const handleAnswer = useCallback((index: number) => {
  1028→    const state = gs.current;
  1029→    if (!state || !state.currentQuiz || state.currentQuiz.answered || state.phase !== 'playing') return;
  1030→    const quiz = state.currentQuiz;
  1031→    quiz.answered = true;
  1032→    const isCorrect = index === quiz.correctIndex;
  1033→    quiz.wasCorrect = isCorrect;
  1034→    state.wordsAnswered++;
  1035→    const now = Date.now();
  1036→    const { w, h } = dims.current;
  1037→    if (isCorrect) {
  1038→      state.wordsCorrect++;
  1039→      state.comboCount++;
  1040→      const reward = QUIZ_SUN_REWARD[quiz.word.difficulty];
  1041→      const comboBonus = state.comboCount >= 3 ? Math.floor(reward * 0.5) : 0;
  1042→      state.sun += reward + comboBonus;
  1043→      state.score += reward + comboBonus;
  1044→      state.floatingTexts.push({
  1045→        id: uid(), x: w / 2, y: h - 100,
  1046→        text: `+${reward + comboBonus} ${comboBonus > 0 ? ' x' + state.comboCount + '连击!' : ''}`,
  1047→        color: '#FF8F00', timer: 1500, maxTimer: 1500,
