    70→const shuffle = <T,>(arr: T[]): T[] => {
    71→  const a = [...arr];
    72→  for (let i = a.length - 1; i > 0; i--) {
    73→    const j = Math.floor(Math.random() * (i + 1));
    74→    [a[i], a[j]] = [a[j], a[i]];
    75→  }
    76→  return a;
    77→};

   499→function drawZombie(ctx: CanvasRenderingContext2D, zombie: Zombie, cellW: number, cellH: number, ox: number, oy: number) {
   500→  const zy = oy + zombie.row * cellH + cellH / 2;
   501→  const bob = zombie.eating ? 0 : Math.sin(zombie.animPhase) * 3;
   502→  const alpha = zombie.dead ? Math.max(0, zombie.deathTimer / 500) : 1;
   503→  ctx.globalAlpha = alpha;
   504→  // Shadow
   505→  ctx.fillStyle = 'rgba(0,0,0,0.15)';
   506→  ctx.beginPath(); ctx.ellipse(zombie.x, oy + zombie.row * cellH + cellH - 6, cellW * 0.2, 5, 0, 0, Math.PI * 2); ctx.fill();
   507→  const { s, bob: b } = drawZombieBase(ctx, zombie.x, zy, bob, cellW, cellH, zombie.animPhase, zombie.slowed, zombie.eating);
   508→
   509→  // Accessories (bigger, more detailed)
   510→  if (zombie.type === 'cone') {
   511→    const cg = ctx.createLinearGradient(zombie.x - s * 0.14, zy - s * 0.6 + b, zombie.x + s * 0.14, zy - s * 0.25 + b);
   512→    cg.addColorStop(0, '#FFB74D'); cg.addColorStop(1, '#E65100');
   513→    ctx.fillStyle = cg;
   514→    ctx.beginPath(); ctx.moveTo(zombie.x, zy - s * 0.62 + b);

   516→    ctx.fillStyle = 'rgba(255,255,255,0.5)';
   517→    ctx.fillRect(zombie.x - s * 0.1, zy - s * 0.46 + b, s * 0.2, s * 0.05);
   518→    ctx.fillRect(zombie.x - s * 0.07, zy - s * 0.37 + b, s * 0.14, s * 0.04);
   519→  } else if (zombie.type === 'bucket') {
   520→    const bg = ctx.createLinearGradient(zombie.x - s * 0.16, zy - s * 0.55 + b, zombie.x + s * 0.16, zy - s * 0.25 + b);
   521→    bg.addColorStop(0, '#B0BEC5'); bg.addColorStop(0.4, '#78909C'); bg.addColorStop(0.7, '#607D8B'); bg.addColorStop(1, '#455A64');
   522→    ctx.fillStyle = bg;
   523→    ctx.beginPath(); ctx.roundRect(zombie.x - s * 0.16, zy - s * 0.55 + b, s * 0.32, s * 0.32, 3); ctx.fill();
   524→    ctx.fillStyle = '#CFD8DC'; ctx.fillRect(zombie.x - s * 0.17, zy - s * 0.55 + b, s * 0.34, s * 0.05);
   525→    ctx.strokeStyle = '#90A4AE'; ctx.lineWidth = 2.5;
   526→    ctx.beginPath(); ctx.arc(zombie.x, zy - s * 0.55 + b, s * 0.1, Math.PI, 0); ctx.stroke();
   527→    ctx.fillStyle = '#CFD8DC';
   528→    ctx.beginPath(); ctx.arc(zombie.x - s * 0.1, zy - s * 0.38 + b, s * 0.02, 0, Math.PI * 2); ctx.fill();
   529→    ctx.beginPath(); ctx.arc(zombie.x + s * 0.1, zy - s * 0.38 + b, s * 0.02, 0, Math.PI * 2); ctx.fill();
   530→  } else if (zombie.type === 'flag') {
   531→    ctx.strokeStyle = '#5D4037'; ctx.lineWidth = 3;
   532→    ctx.beginPath(); ctx.moveTo(zombie.x + s * 0.14, zy - s * 0.18 + b);
   533→    ctx.lineTo(zombie.x + s * 0.14, zy - s * 0.65 + b); ctx.stroke();
   534→    const wave = Math.sin(zombie.animPhase * 2) * s * 0.04;
   535→    ctx.fillStyle = '#E53935';
   536→    ctx.beginPath();
   537→    ctx.moveTo(zombie.x + s * 0.14, zy - s * 0.65 + b);

   539→    ctx.lineTo(zombie.x + s * 0.14, zy - s * 0.42 + b);
   540→    ctx.closePath(); ctx.fill();
   541→  }
   542→
   543→  if (zombie.slowed && !zombie.dead) {
   544→    ctx.fillStyle = 'rgba(100,220,255,0.1)';
   545→    ctx.beginPath(); ctx.arc(zombie.x, zy + bob, s * 0.35, 0, Math.PI * 2); ctx.fill();
   546→  }
   547→  ctx.globalAlpha = 1;
   548→}
