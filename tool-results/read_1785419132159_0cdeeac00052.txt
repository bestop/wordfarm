    70→const shuffle = <T,>(arr: T[]): T[] => {
    71→  const a = [...arr];
    72→  for (let i = a.length - 1; i > 0; i--) {
    73→    const j = Math.floor(Math.random() * (i + 1));
    74→    [a[i], a[j]] = [a[j], a[i]];
    75→  }
    76→  return a;
    77→};

   127→function drawWallnut(ctx: CanvasRenderingContext2D, x: number, y: number, s: number, anim: number, hpRatio: number) {
   128→  const bob = Math.sin(anim * 0.8) * 1.5;
   129→  const squish = 1 + Math.sin(anim * 1.2) * 0.02;
   130→  // Main body
   131→  const wg = ctx.createRadialGradient(x - s * 0.1, y - s * 0.1 + bob, s * 0.1, x, y + bob, s * 0.42 * squish);
   132→  wg.addColorStop(0, '#D4A34A');
   133→  wg.addColorStop(0.6, '#A0722A');
   134→  wg.addColorStop(1, '#6B4513');
   135→  ctx.fillStyle = wg;
   136→  ctx.beginPath();
   137→  ctx.ellipse(x, y + s * 0.05 + bob, s * 0.35 * squish, s * 0.42, 0, 0, Math.PI * 2);
   138→  ctx.fill();
   139→  // Texture lines
   140→  ctx.strokeStyle = 'rgba(90,50,10,0.3)';
   141→  ctx.lineWidth = 1;
   142→  ctx.beginPath(); ctx.arc(x, y + bob, s * 0.2, 0.3, 1.2); ctx.stroke();
   143→  ctx.beginPath(); ctx.arc(x + s * 0.1, y + s * 0.15 + bob, s * 0.15, -0.5, 0.8); ctx.stroke();
   144→  // Cracks when damaged
   145→  if (hpRatio < 0.66) {
   146→    ctx.strokeStyle = 'rgba(50,30,5,0.6)';
   147→    ctx.lineWidth = 1.5;
   148→    ctx.beginPath();
   149→    ctx.moveTo(x - s * 0.1, y - s * 0.15 + bob);
   150→    ctx.lineTo(x - s * 0.05, y + s * 0.05 + bob);
   151→    ctx.lineTo(x - s * 0.15, y + s * 0.2 + bob);
   152→    ctx.stroke();
   153→  }
   154→  if (hpRatio < 0.33) {
   155→    ctx.beginPath();
   156→    ctx.moveTo(x + s * 0.15, y - s * 0.2 + bob);
   157→    ctx.lineTo(x + s * 0.08, y + bob);
   158→    ctx.lineTo(x + s * 0.18, y + s * 0.15 + bob);
   159→    ctx.stroke();
   160→    ctx.beginPath();
   161→    ctx.moveTo(x - s * 0.05, y + s * 0.1 + bob);
   162→    ctx.lineTo(x + s * 0.05, y + s * 0.25 + bob);
   163→    ctx.stroke();
   164→  }
   165→  // Eyes
   166→  ctx.fillStyle = '#fff';
   167→  ctx.beginPath(); ctx.ellipse(x - s * 0.1, y - s * 0.08 + bob, s * 0.07, s * 0.08, 0, 0, Math.PI * 2); ctx.fill();
   168→  ctx.beginPath(); ctx.ellipse(x + s * 0.1, y - s * 0.08 + bob, s * 0.07, s * 0.08, 0, 0, Math.PI * 2); ctx.fill();
   169→  ctx.fillStyle = '#2c1810';
   170→  ctx.beginPath(); ctx.arc(x - s * 0.08, y - s * 0.07 + bob, s * 0.035, 0, Math.PI * 2); ctx.fill();
   171→  ctx.beginPath(); ctx.arc(x + s * 0.12, y - s * 0.07 + bob, s * 0.035, 0, Math.PI * 2); ctx.fill();
   172→  // Worried mouth
   173→  ctx.strokeStyle = '#5D3A1A';
   174→  ctx.lineWidth = 1.5;
   175→  ctx.beginPath();
   176→  ctx.arc(x, y + s * 0.08 + bob, s * 0.08, 0.2, Math.PI - 0.2);
   177→  ctx.stroke();
   178→  // Highlight

   231→  // Eyes

   233→  ctx.beginPath(); ctx.ellipse(x - s * 0.1, y - s * 0.22 + bob, s * 0.1, s * 0.11, 0, 0, Math.PI * 2); ctx.fill();

   248→  }

   402→  ctx.restore();

   418→  ctx.lineTo(x + s * 0.12, zy + s * 0.15 + bob);

   425→  ctx.moveTo(x, zy - s * 0.02 + bob);

   448→  ctx.beginPath(); ctx.arc(x + s * 0.38, zy - s * 0.18 + bob - armSwing, s * 0.05, 0, Math.PI * 2); ctx.fill();

   450→  // Head

   471→  ctx.fillStyle = '#2c1a0a';

   473→  ctx.ellipse(x, zy - s * 0.08 + bob, s * 0.08, s * 0.04, 0, 0, Math.PI * 2);

   480→    ctx.fill();

   482→
   483→  return { s, bob, skinColor };

   486→function drawZombie(ctx: CanvasRenderingContext2D, zombie: Zombie, cellW: number, cellH: number,
   487→  ox: number, oy: number) {
   488→  const zy = oy + zombie.row * cellH + cellH / 2;
   489→  const bob = zombie.eating ? 0 : Math.sin(zombie.animPhase) * 3;
   490→  const alpha = zombie.dead ? Math.max(0, zombie.deathTimer / 500) : 1;
   491→  ctx.globalAlpha = alpha;
   492→
   493→  // Shadow
   494→  ctx.fillStyle = 'rgba(0,0,0,0.2)';
   495→  ctx.beginPath();
   496→  ctx.ellipse(zombie.x, oy + zombie.row * cellH + cellH - 6, cellW * 0.18, 5, 0, 0, Math.PI * 2);
   497→  ctx.fill();
   498→
   499→  const { s, bob: b } = drawZombieBase(ctx, zombie.x, zy, bob, cellW, cellH, zombie.animPhase, zombie.slowed, zombie.eating);
   500→
   501→  // Accessories
   502→  if (zombie.type === 'cone') {
   503→    // Orange traffic cone
   504→    const cg = ctx.createLinearGradient(zombie.x - s * 0.12, zy - s * 0.5 + b, zombie.x + s * 0.12, zy - s * 0.2 + b);
   505→    cg.addColorStop(0, '#FF9800');
   506→    cg.addColorStop(1, '#E65100');

   509→    ctx.moveTo(zombie.x, zy - s * 0.52 + b);
   510→    ctx.lineTo(zombie.x - s * 0.14, zy - s * 0.22 + b);
   511→    ctx.lineTo(zombie.x + s * 0.14, zy - s * 0.22 + b);
   512→    ctx.closePath();
   513→    ctx.fill();
   514→    // White stripes
   515→    ctx.fillStyle = 'rgba(255,255,255,0.6)';
   516→    ctx.fillRect(zombie.x - s * 0.09, zy - s * 0.38 + b, s * 0.18, s * 0.04);
   517→    ctx.fillRect(zombie.x - s * 0.06, zy - s * 0.3 + b, s * 0.12, s * 0.03);
   518→  } else if (zombie.type === 'bucket') {
   519→    // Metal bucket
   520→    const bg = ctx.createLinearGradient(zombie.x - s * 0.15, zy - s * 0.48 + b, zombie.x + s * 0.15, zy - s * 0.18 + b);
   521→    bg.addColorStop(0, '#90A4AE');
   522→    bg.addColorStop(0.3, '#78909C');
   523→    bg.addColorStop(0.7, '#607D8B');
   524→    bg.addColorStop(1, '#455A64');
   525→    ctx.fillStyle = bg;
   526→    ctx.beginPath();
   527→    ctx.roundRect(zombie.x - s * 0.14, zy - s * 0.45 + b, s * 0.28, s * 0.28, 3);
   528→    ctx.fill();
   529→    // Bucket rim
   530→    ctx.fillStyle = '#B0BEC5';
   531→    ctx.fillRect(zombie.x - s * 0.15, zy - s * 0.45 + b, s * 0.3, s * 0.04);
   532→    // Handle
   533→    ctx.strokeStyle = '#90A4AE'; ctx.lineWidth = 2;
   534→    ctx.beginPath();
   535→    ctx.arc(zombie.x, zy - s * 0.45 + b, s * 0.1, Math.PI, 0);
   536→    ctx.stroke();
   537→    // Rivets
   538→    ctx.fillStyle = '#CFD8DC';
   539→    ctx.beginPath(); ctx.arc(zombie.x - s * 0.1, zy - s * 0.3 + b, s * 0.02, 0, Math.PI * 2); ctx.fill();
   540→    ctx.beginPath(); ctx.arc(zombie.x + s * 0.1, zy - s * 0.3 + b, s * 0.02, 0, Math.PI * 2); ctx.fill();
   541→  } else if (zombie.type === 'flag') {
   542→    // Flag pole
   543→    ctx.strokeStyle = '#5D4037'; ctx.lineWidth = 2.5;
   544→    ctx.beginPath();
   545→    ctx.moveTo(zombie.x + s * 0.12, zy - s * 0.15 + b);
   546→    ctx.lineTo(zombie.x + s * 0.12, zy - s * 0.6 + b);
   547→    ctx.stroke();
   548→    // Flag (waving)
   549→    const wave = Math.sin(zombie.animPhase * 2) * s * 0.03;
   550→    ctx.fillStyle = '#D32F2F';
   551→    ctx.beginPath();
   552→    ctx.moveTo(zombie.x + s * 0.12, zy - s * 0.6 + b);
   553→    ctx.quadraticCurveTo(zombie.x + s * 0.25 + wave, zy - s * 0.55 + b, zombie.x + s * 0.32, zy - s * 0.5 + b);
   554→    ctx.lineTo(zombie.x + s * 0.12, zy - s * 0.4 + b);
   555→    ctx.closePath();
   556→    ctx.fill();
   557→    // Skull on flag
   558→    ctx.fillStyle = '#fff';
   559→    ctx.font = `${s * 0.1}px sans-serif`;
   560→    ctx.textAlign = 'center';
   561→    ctx.fillText('☠', zombie.x + s * 0.2, zy - s * 0.48 + b);
   562→  }
   563→
   564→  // Slowed aura
   565→  if (zombie.slowed && !zombie.dead) {
   566→    ctx.fillStyle = 'rgba(0, 188, 212, 0.15)';
   567→    ctx.beginPath(); ctx.arc(zombie.x, zy + bob, s * 0.35, 0, Math.PI * 2); ctx.fill();
   568→    ctx.strokeStyle = 'rgba(0, 229, 255, 0.3)';
   569→    ctx.lineWidth = 1;
   570→    ctx.stroke();
   571→  }
   572→
   573→  ctx.globalAlpha = 1;
   574→}

   577→export default function PvZGame() {
   578→  const canvasRef = useRef<HTMLCanvasElement>(null);
   579→  const containerRef = useRef<HTMLDivElement>(null);
   580→  const gs = useRef<GameState | null>(null);
   581→  const rafRef = useRef<number>(0);

   583→  const [uiTick, setUiTick] = useState(0);
   584→  const forceUpdate = useCallback(() => setUiTick(t => t + 1), []);
   585→
   586→  const dims = useRef({ w: 900, h: 500, cellW: 80, cellH: 90, ox: 30, oy: 15 });
   587→
   588→  const recalcDims = useCallback(() => {
   589→    const canvas = canvasRef.current;
   590→    const container = containerRef.current;
   591→    if (!canvas || !container) return;
   592→    const rect = container.getBoundingClientRect();
   593→    const dpr = window.devicePixelRatio || 1;
   594→    const w = rect.width;
   595→    const h = rect.height;
   596→    canvas.width = w * dpr;
   597→    canvas.height = h * dpr;
   598→    canvas.style.width = `${w}px`;
   599→    canvas.style.height = `${h}px`;
   600→    const ctx = canvas.getContext('2d');
   601→    if (ctx) ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   602→    const cellW = (w * 0.9) / GRID_COLS;
   603→    const cellH = (h * 0.95) / GRID_ROWS;
   604→    const ox = w * 0.07;
   605→    const oy = h * 0.025;
   606→    dims.current = { w, h, cellW, cellH, ox, oy };
   607→  }, []);
   608→
   609→  const initGame = useCallback(() => {
   610→    _idCounter = 0;
   611→    const state: GameState = {
   612→      phase: 'playing', sun: STARTING_SUN, score: 0, wave: 0,
   613→      plants: [], zombies: [], projectiles: [],
   614→      explosions: [], floatingTexts: [],
   615→      selectedPlant: null, currentQuiz: null, quizCooldown: 0,
   616→      wordsAnswered: 0, wordsCorrect: 0,
   617→      waveStartTime: Date.now(), waveZombiesSpawned: 0,
   618→      totalKills: 0, lastTime: Date.now(),
   619→      zombieSpeedBoostEnd: 0, usedWordIndices: new Set(),
   620→      shakeTimer: 0, comboCount: 0,
   621→    };

   623→    setPhase('playing');

   625→    forceUpdate();
   626→  }, [forceUpdate]);
   627→
   628→  const generateQuiz = useCallback((state: GameState) => {
   629→    const difficultyFilter = state.wordsAnswered < 3 ? 1 : 0;
   630→    let available = WORD_BANK.map((_, i) => i).filter(
   631→      i => !state.usedWordIndices.has(i) && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
   632→    );
   633→    if (available.length < 4) { state.usedWordIndices.clear(); }
   634→    available = WORD_BANK.map((_, i) => i).filter(
   635→      i => !state.usedWordIndices.has(i) && (!difficultyFilter || WORD_BANK[i].difficulty === 1)
   636→    );
   637→    if (available.length < 4) available = WORD_BANK.map((_, i) => i);
   638→    const shuffled = shuffle(available);
   639→    const correctIdx = shuffled[0];
   640→    const correctWord = WORD_BANK[correctIdx];
   641→    state.usedWordIndices.add(correctIdx);
   642→    const wrongPool = WORD_BANK.map((_, i) => i).filter( i => i !== correctIdx && (!difficultyFilter || WORD_BANK[i].difficulty === 1) );
   643→    const wrongShuffled = shuffle(wrongPool).slice(0, 3);
   644→    const options = shuffle([correctWord.zh, ...wrongShuffled.map(i => WORD_BANK[i].zh)]);
   645→    state.currentQuiz = {
   646→      word: correctWord, options,
   647→      correctIndex: options.indexOf(correctWord.zh),
   648→      timer: QUIZ_TIME_LIMIT, answered: false, wasCorrect: undefined,
   649→    };
   650→    forceUpdate();
   651→  }, [forceUpdate]);
   652→
   653→  const handleAnswer = useCallback((index: number) => {
   654→    const state = gs.current;
   655→    if (!state || !state.currentQuiz || state.currentQuiz.answered || state.phase !== 'playing') return;
   656→    const quiz = state.currentQuiz;
   657→    quiz.answered = true;
   658→    const isCorrect = index === quiz.correctIndex;
   659→    quiz.wasCorrect = isCorrect;
   660→    state.wordsAnswered++;
   661→    const now = Date.now();
   662→    const { w, h } = dims.current;
   663→    if (isCorrect) {
   664→      state.wordsCorrect++;
   665→      state.comboCount++;
   666→      const reward = QUIZ_SUN_REWARD[quiz.word.difficulty];
   667→      const comboBonus = state.comboCount >= 3 ? Math.floor(reward * 0.5) : 0;
   668→      state.sun += reward + comboBonus;
   669→      state.score += reward + comboBonus;
   670→      state.floatingTexts.push({
   671→        id: uid(), x: w / 2, y: h - 80,
   672→        text: `+${reward + comboBonus} ☀️${comboBonus > 0 ? ' 连击x' + state.comboCount : ''}`,
   673→        color: '#FFD700', timer: 1500, maxTimer: 1500,
   674→      });
   675→    } else {
   676→      state.comboCount = 0;
   677→      state.zombieSpeedBoostEnd = now + ZOMBIE_SPEED_BOOST_DURATION;
   678→      state.shakeTimer = 300;
   679→      state.floatingTexts.push({
   680→        id: uid(), x: w / 2, y: h - 80,
   681→        text: '答错了! 僵尸加速!', color: '#FF4444', timer: 1500, maxTimer: 1500,
   682→      });
   683→    }
   684→    state.quizCooldown = QUIZ_COOLDOWN;
   685→    forceUpdate();
   686→  }, [forceUpdate]);
   687→
   688→  const handleCanvasClick = useCallback((e: React.MouseEvent | React.TouchEvent) => {
   689→    const state = gs.current;
   690→    if (!state || state.phase !== 'playing' || !state.selectedPlant) return;
   691→    const canvas = canvasRef.current;
   692→    if (!canvas) return;
   693→    const rect = canvas.getBoundingClientRect();
   694→    let clientX: number, clientY: number;
   695→    if ('touches' in e) {
   696→      if (e.touches.length === 0) return;
   697→      clientX = e.touches[0].clientX;
   698→      clientY = e.touches[0].clientY;
   699→    } else {
   700→      clientX = e.clientX;
   701→      clientY = e.clientY;
   702→    }
   703→    const x = clientX - rect.left;
   704→    const y = clientY - rect.top;
   705→    const { cellW, cellH, ox, oy } = dims.current;
   706→    const col = Math.floor((x - ox) / cellW);
   707→    const row = Math.floor((y - oy) / cellH);
   708→    if (col < 0 || col >= GRID_COLS || row < 0 || row >= GRID_ROWS) return;
   709→    if (state.plants.some(p => p.row === row && p.col === col)) return;
   710→    const def = PLANT_DEFS[state.selectedPlant];
   711→    if (state.sun < def.cost) {
   712→      state.floatingTexts.push({
   713→        id: uid(), x, y, text: '阳光不足!', color: '#FF4444', timer: 1000, maxTimer: 1000,
   714→      });
   715→      return;
   716→    }
   717→    state.sun -= def.cost;
   718→    const plant: Plant = {
   719→      id: uid(), type: state.selectedPlant, row, col,
   720→      hp: def.hp, maxHp: def.hp,
   721→      lastAttack: Date.now(), animPhase: Math.random() * Math.PI * 2,
   722→    };
   723→    state.plants.push(plant);
   724→
   725→    if (def.explosive) {
   726→      const capturedState = state;
   727→      const capturedPlant = plant;
   728→      const capturedCol = col;
   729→      const capturedRow = row;
   730→      setTimeout(() => {
   731→        if (capturedState.phase !== 'playing') return;
   732→        const { ox: oX, oy: oY, cellW: cW, cellH: cH } = dims.current;
   733→        const cx = oX + capturedCol * cW + cW / 2;
   734→        const cy = oY + capturedRow * cH + cH / 2;
   735→        capturedState.explosions.push({
   736→          id: uid(), x: cx, y: cy, radius: 0,
   737→          maxRadius: cW * 2, timer: 0, maxTimer: 600,
   738→        });
   739→        capturedState.zombies.forEach(z => {
   740→          if (!z.dead && Math.abs(z.row - capturedRow) <= 1) {
   741→            const zy = oY + z.row * cH + cH / 2;
   742→            const dist = Math.sqrt((z.x - cx) ** 2 + (zy - cy) ** 2);
   743→            if (dist < cW * 2.5) {
   744→              z.hp -= 1800;
   745→              if (z.hp <= 0) { z.dead = true; z.deathTimer = 500; capturedState.totalKills++; capturedState.score += 50; }
   746→            }
   747→          }
   748→        });
   749→        capturedState.plants = capturedState.plants.filter(p => p.id !== capturedPlant.id);
   750→        forceUpdate();
   751→      }, 500);
   752→    }
   753→    forceUpdate();
   754→  }, [forceUpdate]);
   755→
   756→  // ---- Game loop ----
   757→  const gameLoop = useCallback(() => {
   758→    const state = gs.current;
   759→    const canvas = canvasRef.current;
   760→    if (!state || !canvas) return;
   761→    if (state.phase !== 'playing') return;
   762→    const ctx = canvas.getContext('2d');
   763→    if (!ctx) return;
   764→
   765→    const now = Date.now();
   766→    const dt = Math.min(now - state.lastTime, 50);
   767→    state.lastTime = now;
   768→    const { w, h, cellW, cellH, ox, oy } = dims.current;
   769→    const isBoosted = now < state.zombieSpeedBoostEnd;
   770→    const speedMult = isBoosted ? ZOMBIE_SPEED_BOOST : 1;
   771→
   772→    // -- Spawn wave zombies --
   773→    const waveConfig = WAVE_CONFIGS[state.wave];
   774→    if (waveConfig) {
   775→      const waveElapsed = now - state.waveStartTime;
   776→      for (let i = state.waveZombiesSpawned; i < waveConfig.zombies.length; i++) {
   777→        if (waveElapsed >= waveConfig.zombies[i].delay) {
   778→          state.waveZombiesSpawned = i + 1;
   779→          const zc = waveConfig.zombies[i];
   780→          const def = ZOMBIE_DEFS[zc.type];
   781→          const row = zc.row ?? Math.floor(Math.random() * GRID_ROWS);
   782→          state.zombies.push({
   783→            id: uid(), type: zc.type, row,
   784→            x: ox + (GRID_COLS + 0.8) * cellW,
   785→            hp: def.hp, maxHp: def.hp, speed: def.speed, baseSpeed: def.speed,
   786→            eating: false, slowed: false, slowTimer: 0,
   787→            lastHit: now, animPhase: Math.random() * Math.PI * 2,
   788→            dead: false, deathTimer: 0,
   789→          });
   790→        } else break;
   791→      }
   792→    }
   793→
   794→    // -- Check wave complete --
   795→    const allSpawned = waveConfig ? state.waveZombiesSpawned >= waveConfig.zombies.length : true;
   796→    const allDead = state.zombies.length === 0 || state.zombies.every(z => z.dead);
   797→    if (allSpawned && allDead && state.zombies.length > 0) {
   798→      if (state.wave < WAVE_CONFIGS.length - 1) {
   799→        state.wave++;
   800→        state.waveStartTime = now;
   801→        state.waveZombiesSpawned = 0;
   802→        state.floatingTexts.push({
   803→          id: uid(), x: w / 2, y: h / 2,
   804→          text: `第 ${state.wave + 1} 波!`, color: '#FF6600', timer: 2000, maxTimer: 2000,
   805→        });
   806→        forceUpdate();
   807→      } else {
   808→        state.phase = 'victory';
   809→        setPhase('victory');
   810→        return;
   811→      }
   812→    }
   813→
   814→    // -- Update zombies --
   815→    let gameOver = false;
   816→    for (const zombie of state.zombies) {
   817→      if (zombie.dead) { zombie.deathTimer -= dt; continue; }
   818→      if (zombie.slowed) { zombie.slowTimer -= dt; if (zombie.slowTimer <= 0) zombie.slowed = false; }
   819→      zombie.animPhase += dt * 0.005 * (zombie.slowed ? 0.5 : 1);
   820→      const currentSpeed = zombie.baseSpeed * speedMult * (zombie.slowed ? 0.5 : 1);
   821→      const eatingPlant = state.plants.find(p => {
   822→        if (p.row !== zombie.row) return false;
   823→        const plantX = ox + p.col * cellW + cellW / 2;
   824→        return Math.abs(zombie.x - plantX) < cellW * 0.35;
   825→      });
   826→      if (eatingPlant) {
   827→        zombie.eating = true;
   828→        if (now - zombie.lastHit > 1000) {
   829→          zombie.lastHit = now;
   830→          eatingPlant.hp -= 100;
   831→          if (eatingPlant.hp <= 0) state.plants = state.plants.filter(p => p.id !== eatingPlant.id);
   832→        }
   833→      } else {
   834→        zombie.eating = false;
   835→        zombie.x -= currentSpeed * (dt / 1000);
   836→      }
   837→      if (zombie.x < ox - cellW * 0.5) gameOver = true;
   838→    }
   839→    state.zombies = state.zombies.filter(z => !(z.dead && z.deathTimer <= 0));
   840→
   841→    if (gameOver) { state.phase = 'gameover'; setPhase('gameover'); return; }
   842→
   843→    // -- Update plants --
   844→    for (const plant of state.plants) {
   845→      plant.animPhase += dt * 0.003;
   846→      const def = PLANT_DEFS[plant.type];
   847→      if (def.attack && def.attackSpeed) {
   848→        const hasZombie = state.zombies.some(z => !z.dead && z.row === plant.row && z.x > ox + plant.col * cellW);
   849→        if (hasZombie && now - plant.lastAttack >= def.attackSpeed) {
   850→          plant.lastAttack = now;
   851→          const px = ox + plant.col * cellW + cellW * 0.7;
   852→          state.projectiles.push({ id: uid(), row: plant.row, x: px, speed: 250, damage: def.attack, slow: !!def.slowEffect, active: true });
   853→          if (def.doubleShot) {
   854→            setTimeout(() => {
   855→              if (gs.current?.phase === 'playing') {
   856→                gs.current.projectiles.push({ id: uid(), row: plant.row, x: px, speed: 250, damage: def.attack, slow: false, active: true });
   857→              }
   858→            }, 150);
   859→          }
   860→        }
   861→      }
   862→    }
   863→
   864→    // -- Update projectiles --
   865→    for (const proj of state.projectiles) {
   866→      if (!proj.active) continue;
   867→      proj.x += proj.speed * (dt / 1000);
   868→      const hitZombie = state.zombies.find(z => !z.dead && z.row === proj.row && Math.abs(z.x - proj.x) < cellW * 0.25);
   869→      if (hitZombie) {
   870→        proj.active = false;
   871→        hitZombie.hp -= proj.damage;
   872→        if (proj.slow) { hitZombie.slowed = true; hitZombie.slowTimer = 3000; }
   873→        if (hitZombie.hp <= 0) { hitZombie.dead = true; hitZombie.deathTimer = 500; state.totalKills++; state.score += 50; }
   874→        state.floatingTexts.push({
   875→          id: uid(), x: proj.x, y: oy + proj.row * cellH + cellH * 0.25,
   876→          text: `-${proj.damage}`, color: proj.slow ? '#00E5FF' : '#FF6600', timer: 600, maxTimer: 600,
   877→        });
   878→      }
   879→      if (proj.x > ox + (GRID_COLS + 1) * cellW) proj.active = false;
   880→    }
   881→    state.projectiles = state.projectiles.filter(p => p.active);
   882→
   883→    // -- Update effects --
   884→    for (const exp of state.explosions) { exp.timer += dt; exp.radius = (exp.timer / exp.maxTimer) * exp.maxRadius; }
   885→    state.explosions = state.explosions.filter(e => e.timer < e.maxTimer);
   886→    for (const ft of state.floatingTexts) { ft.timer -= dt; ft.y -= dt * 0.04; }
   887→    state.floatingTexts = state.floatingTexts.filter(ft => ft.timer > 0);
   888→
   889→    // -- Update quiz --
   890→    if (state.currentQuiz) {
   891→      if (state.currentQuiz.answered) {
   892→        state.quizCooldown -= dt;
   893→        if (state.quizCooldown <= 0) generateQuiz(state);
   894→      } else {
   895→        state.currentQuiz.timer -= dt;
   896→        if (state.currentQuiz.timer <= 0) {
   897→          state.currentQuiz.answered = true;
   898→          state.currentQuiz.wasCorrect = false;
   899→          state.wordsAnswered++;
   900→          state.comboCount = 0;
   901→          state.zombieSpeedBoostEnd = now + ZOMBIE_SPEED_BOOST_DURATION;
   902→          state.shakeTimer = 300;
   903→          state.quizCooldown = QUIZ_COOLDOWN;
   904→          state.floatingTexts.push({
   905→            id: uid(), x: w / 2, y: h - 80,
   906→            text: '⏰ 超时! 僵尸加速!', color: '#FF4444', timer: 1500, maxTimer: 1500,
   907→          });
   908→          forceUpdate();
   909→        }
   910→      }
   911→    }
   912→
   913→    if (state.shakeTimer > 0) state.shakeTimer -= dt;
   914→
   915→    // ======== RENDER ========
   916→    const dpr = window.devicePixelRatio || 1;
   917→    ctx.save();
   918→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   919→
   920→    if (state.shakeTimer > 0) {
   921→      ctx.translate((Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6);
   922→    }
   923→
   924→    // Background - sky
   925→    const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
   926→    skyGrad.addColorStop(0, '#5BA3D9');
   927→    skyGrad.addColorStop(0.15, '#87CEEB');
   928→    skyGrad.addColorStop(0.3, '#B0E0E6');
   929→    skyGrad.addColorStop(0.45, '#90EE90');
   930→    skyGrad.addColorStop(1, '#2E7D32');
   931→    ctx.fillStyle = skyGrad;
   932→    ctx.fillRect(0, 0, w, h);
   933→
   934→    // Clouds
   935→    ctx.fillStyle = 'rgba(255,255,255,0.5)';
   936→    const cloudY = h * 0.06;
   937→    ctx.beginPath(); ctx.ellipse(w * 0.15, cloudY, 40, 14, 0, 0, Math.PI * 2); ctx.fill();
   938→    ctx.beginPath(); ctx.ellipse(w * 0.18, cloudY - 5, 25, 10, 0, 0, Math.PI * 2); ctx.fill();
   939→    ctx.beginPath(); ctx.ellipse(w * 0.55, cloudY + 8, 35, 12, 0, 0, Math.PI * 2); ctx.fill();
   940→    ctx.beginPath(); ctx.ellipse(w * 0.58, cloudY + 3, 28, 10, 0, 0, Math.PI * 2); ctx.fill();
   941→    ctx.beginPath(); ctx.ellipse(w * 0.82, cloudY - 3, 32, 11, 0, 0, Math.PI * 2); ctx.fill();
   942→
   943→    // Grid
   944→    for (let r = 0; r < GRID_ROWS; r++) {
   945→      for (let c = 0; c < GRID_COLS; c++) {
   946→        const isLight = (r + c) % 2 === 0;
   947→        ctx.fillStyle = isLight ? 'rgba(76, 175, 80, 0.35)' : 'rgba(56, 142, 60, 0.3)';
   948→        ctx.fillRect(ox + c * cellW, oy + r * cellH, cellW, cellH);
   949→        // Subtle grass texture
   950→        ctx.fillStyle = isLight ? 'rgba(100, 200, 100, 0.12)' : 'rgba(60, 160, 60, 0.08)';
   951→        for (let gi = 0; gi < 3; gi++) {
   952→          const gx = ox + c * cellW + ((r * 7 + c * 13 + gi * 17) % 10) / 10 * cellW;
   953→          const gy = oy + r * cellH + ((r * 11 + c * 3 + gi * 7) % 8) / 8 * cellH;
   954→          ctx.fillRect(gx, gy, 2, 4);
   955→        }
   956→        ctx.strokeStyle = 'rgba(0, 80, 0, 0.08)';
   957→        ctx.lineWidth = 0.5;
   958→        ctx.strokeRect(ox + c * cellW, oy + r * cellH, cellW, cellH);
   959→      }
   960→    }
   961→
   962→    // House edge
   963→    const houseGrad = ctx.createLinearGradient(0, oy, ox, oy);
   964→    houseGrad.addColorStop(0, '#6D4C2A');
   965→    houseGrad.addColorStop(0.6, '#8B6914');
   966→    houseGrad.addColorStop(1, '#A07830');
   967→    ctx.fillStyle = houseGrad;
   968→    ctx.fillRect(0, oy, ox - 2, GRID_ROWS * cellH);
   969→    // Door
   970→    const doorW = ox * 0.45;
   971→    const doorH = cellH * 0.75;
   972→    ctx.fillStyle = '#4E342E';
   973→    const doorX = (ox - doorW) / 2;
   974→    const doorY = oy + (GRID_ROWS * cellH - doorH) / 2;
   975→    ctx.beginPath();
   976→    ctx.roundRect(doorX, doorY, doorW, doorH, [6, 6, 0, 0]);
   977→    ctx.fill();
   978→    // Door knob
   979→    ctx.fillStyle = '#FFD54F';
   980→    ctx.beginPath(); ctx.arc(doorX + doorW * 0.75, doorY + doorH * 0.55, 3, 0, Math.PI * 2); ctx.fill();
   981→    // Windows
   982→    ctx.fillStyle = '#BBDEFB';
   983→    const winS = ox * 0.28;
   984→    ctx.fillRect(ox * 0.15, oy + cellH * 0.3, winS, winS);
   985→    ctx.fillRect(ox * 0.15, oy + cellH * 2.2, winS, winS);
   986→    // Window frames
   987→    ctx.strokeStyle = '#5D4037'; ctx.lineWidth = 1.5;
   988→    ctx.strokeRect(ox * 0.15, oy + cellH * 0.3, winS, winS);
   989→    ctx.strokeRect(ox * 0.15, oy + cellH * 2.2, winS, winS);
   990→    ctx.beginPath();
   991→    ctx.moveTo(ox * 0.15 + winS / 2, oy + cellH * 0.3); ctx.lineTo(ox * 0.15 + winS / 2, oy + cellH * 0.3 + winS);
   992→    ctx.stroke();
   993→    ctx.beginPath();
   994→    ctx.moveTo(ox * 0.15, oy + cellH * 0.3 + winS / 2); ctx.lineTo(ox * 0.15 + winS, oy + cellH * 0.3 + winS / 2);
   995→    ctx.stroke();
   996→
   997→    // Grid hover when plant selected
   998→    if (state.selectedPlant) {
   999→      ctx.fillStyle = 'rgba(255, 255, 0, 0.08)';
  1000→      for (let r = 0; r < GRID_ROWS; r++) {
  1001→        for (let c = 0; c < GRID_COLS; c++) {
  1002→          if (!state.plants.some(p => p.row === r && p.col === c)) {
  1003→            ctx.fillRect(ox + c * cellW, oy + r * cellH, cellW, cellH);
  1004→          }
  1005→        }
  1006→      }
  1007→    }
  1008→
  1009→    // Draw plants
  1010→    ctx.textAlign = 'center';
  1011→    ctx.textBaseline = 'middle';
  1012→    for (const plant of state.plants) {
  1013→      const px = ox + plant.col * cellW + cellW / 2;
  1014→      const py = oy + plant.row * cellH + cellH / 2;
  1015→      const hpRatio = plant.hp / plant.maxHp;
  1016→      // Shadow
  1017→      ctx.fillStyle = 'rgba(0,0,0,0.12)';
  1018→      ctx.beginPath();
  1019→      ctx.ellipse(px, oy + plant.row * cellH + cellH - 5, cellW * 0.22, 5, 0, 0, Math.PI * 2);
  1020→      ctx.fill();
  1021→      drawPlant(ctx, plant.type, px, py, cellW, cellH, plant.animPhase, hpRatio);
  1022→      // HP bar
  1023→      if (plant.hp < plant.maxHp) {
  1024→        const barW = cellW * 0.6;
  1025→        const barX = px - barW / 2;
  1026→        const barY = oy + plant.row * cellH + 3;
  1027→        ctx.fillStyle = 'rgba(0,0,0,0.5)';
  1028→        ctx.beginPath(); ctx.roundRect(barX - 1, barY - 1, barW + 2, 6, 3); ctx.fill();
  1029→        const ratio = Math.max(0, hpRatio);
  1030→        const hpColor = ratio > 0.5 ? '#4CAF50' : ratio > 0.25 ? '#FF9800' : '#F44336';
  1031→        ctx.fillStyle = hpColor;
  1032→        ctx.beginPath(); ctx.roundRect(barX, barY, barW * ratio, 4, 2); ctx.fill();
  1033→      }
  1034→    }
  1035→
  1036→    // Draw projectiles
  1037→    for (const proj of state.projectiles) {
  1038→      const py = oy + proj.row * cellH + cellH / 2;
  1039→      const pSize = Math.max(4, cellW * 0.08);
  1040→      // Trail
  1041→      ctx.fillStyle = proj.slow ? 'rgba(0,229,255,0.3)' : 'rgba(118,255,3,0.3)';
  1042→      ctx.beginPath(); ctx.ellipse(proj.x - pSize * 1.5, py, pSize * 1.8, pSize * 0.8, 0, 0, Math.PI * 2); ctx.fill();
  1043→      // Glow
  1044→      ctx.shadowColor = proj.slow ? '#00E5FF' : '#76FF03';
  1045→      ctx.shadowBlur = 8;
  1046→      // Main projectile
  1047→      const pg = ctx.createRadialGradient(proj.x - pSize * 0.2, py - pSize * 0.2, 0, proj.x, py, pSize);
  1048→      if (proj.slow) {
  1049→        pg.addColorStop(0, '#E0F7FA');
  1050→        pg.addColorStop(0.5, '#00E5FF');
  1051→        pg.addColorStop(1, '#0097A7');
  1052→      } else {
  1053→        pg.addColorStop(0, '#CCFF90');
  1054→        pg.addColorStop(0.5, '#76FF03');
  1055→        pg.addColorStop(1, '#33691E');
  1056→      }
  1057→      ctx.fillStyle = pg;
  1058→      ctx.beginPath(); ctx.arc(proj.x, py, pSize, 0, Math.PI * 2); ctx.fill();
  1059→      ctx.shadowBlur = 0;
  1060→    }
  1061→
  1062→    // Draw zombies
  1063→    for (const zombie of state.zombies) {

  1067→        const barW = cellW * 0.45;
  1068→        const barX = zombie.x - barW / 2;
  1069→        const barY = oy + zombie.row * cellH + 3;
  1070→        ctx.fillStyle = 'rgba(0,0,0,0.5)';
  1071→        ctx.beginPath(); ctx.roundRect(barX - 1, barY - 1, barW + 2, 6, 3); ctx.fill();
  1072→        const ratio = Math.max(0, zombie.hp / zombie.maxHp);
  1073→        const hpColor = ratio > 0.5 ? '#4CAF50' : ratio > 0.25 ? '#FF9800' : '#F44336';
  1074→        ctx.fillStyle = hpColor;
  1075→        ctx.beginPath(); ctx.roundRect(barX, barY, barW * ratio, 4, 2); ctx.fill();
  1076→      }
  1077→    }
  1078→
  1079→    // Explosions
  1080→    for (const exp of state.explosions) {
  1081→      const progress = exp.timer / exp.maxTimer;
  1082→      const alpha = 1 - progress;
  1083→      // Outer ring
  1084→      ctx.strokeStyle = `rgba(255, 200, 0, ${alpha * 0.8})`;
  1085→      ctx.lineWidth = 3;
  1086→      ctx.beginPath(); ctx.arc(exp.x, exp.y, exp.radius, 0, Math.PI * 2); ctx.stroke();
  1087→      // Inner glow
  1088→      ctx.fillStyle = `rgba(255, 100, 0, ${alpha * 0.5})`;
  1089→      ctx.beginPath(); ctx.arc(exp.x, exp.y, exp.radius, 0, Math.PI * 2); ctx.fill();
  1090→      ctx.fillStyle = `rgba(255, 220, 50, ${alpha * 0.4})`;
  1091→      ctx.beginPath(); ctx.arc(exp.x, exp.y, exp.radius * 0.5, 0, Math.PI * 2); ctx.fill();
  1092→      ctx.fillStyle = `rgba(255, 255, 200, ${alpha * 0.6})`;
  1093→      ctx.beginPath(); ctx.arc(exp.x, exp.y, exp.radius * 0.2, 0, Math.PI * 2); ctx.fill();

  1110→
  1111→    // Wave announcement

  1117→        ctx.font = 'bold 30px "Noto Sans SC", sans-serif';

  1119→        ctx.strokeStyle = 'rgba(0,0,0,0.7)'; ctx.lineWidth = 4;
  1120→        ctx.strokeText(`第 ${state.wave + 1} 波`, w / 2, h / 2 - 10);
  1121→        ctx.fillStyle = '#FFFFFF';
  1122→        ctx.fillText(`第 ${state.wave + 1} 波`, w / 2, h / 2 - 10);
  1123→        ctx.font = '16px "Noto Sans SC", sans-serif';
  1124→        ctx.strokeStyle = 'rgba(0,0,0,0.5)'; ctx.lineWidth = 3;
  1125→        ctx.strokeText(`${waveConfig.zombies.length} 个僵尸来袭!`, w / 2, h / 2 + 22);
  1126→        ctx.fillStyle = '#FFE082';
  1127→        ctx.fillText(`${waveConfig.zombies.length} 个僵尸来袭!`, w / 2, h / 2 + 22);

  1139→
  1140→    ctx.restore();
  1141→    rafRef.current = requestAnimationFrame(gameLoop);
  1142→  }, [generateQuiz, forceUpdate]);
  1143→
  1144→  // ---- Effects ----
  1145→  useEffect(() => {

  1147→    recalcDims();
  1148→    const observer = new ResizeObserver(() => recalcDims());
  1149→    observer.observe(containerRef.current);
  1150→    return () => observer.disconnect();
  1151→  }, [phase, recalcDims]);
  1152→
  1153→  useEffect(() => {
  1154→    if (phase === 'playing') {
  1155→      gs.current!.lastTime = Date.now();
  1156→      rafRef.current = requestAnimationFrame(gameLoop);
