    92→const shuffle = <T,>(arr: T[]): T[] => {
    93→  const a = [...arr];
    94→  for (let i = a.length - 1; i > 0; i--) {
    95→    const j = Math.floor(Math.random() * (i + 1));
    96→    [a[i], a[j]] = [a[j], a[i]];
    97→  }
    98→  return a;
    99→};

  1004→function trySpawnDrop(state: GameState, zombie: Zombie, cellW: number) {
  1005→  let dropChance = 0.2;
  1006→  if (zombie.type === 'cone') dropChance = 0.3;
  1007→  if (zombie.type === 'bucket') dropChance = 0.4;
  1008→  if (zombie.type === 'flag') dropChance = 0.55;
  1009→  if (Math.random() < dropChance) {
  1010→    const typeRand = Math.random();
  1011→    let type: Pickup['type'] = 'sun';
  1012→    let value = 25 + Math.floor(Math.random() * 20);
  1013→    if (zombie.type === 'bucket' || zombie.type === 'flag') value = 35 + Math.floor(Math.random() * 25);
  1014→    if (typeRand > 0.85) { type = 'double'; value = 0; }
  1015→    else if (typeRand > 0.7) { type = 'freeze'; value = 0; }
  1016→    state.pickups.push({
  1017→      id: uid(), x: zombie.x, row: zombie.row,
  1018→      type, timer: 8000, maxTimer: 8000, collected: false,
  1019→      bobPhase: Math.random() * Math.PI * 2, value,
  1020→    });
  1021→  }

  1024→function collectPickup(state: GameState, pk: Pickup, cellW: number, cellH: number, ox: number, oy: number) {
  1025→  pk.collected = true;
  1026→  const py = oy + pk.row * cellH + cellH / 2;
  1027→  if (pk.type === 'sun') {
  1028→    state.sun += pk.value;
  1029→    state.score += pk.value;
  1030→    state.floatingTexts.push({ id: uid(), x: pk.x, y: py, text: `+${pk.value}☀️`, color: '#FFD54F', timer: 1000, maxTimer: 1000 });
  1031→  } else if (pk.type === 'double') {
  1032→    state.doubleDamageEnd = Date.now() + 6000;
  1033→    state.floatingTexts.push({ id: uid(), x: pk.x, y: py, text: '⚡双倍伤害 6s!', color: '#FF6F00', timer: 1500, maxTimer: 1500 });
  1034→  } else if (pk.type === 'freeze') {
  1035→    state.zombies.forEach(z => { if (!z.dead) { z.slowed = true; z.slowTimer = 4000; } });
  1036→    state.floatingTexts.push({ id: uid(), x: pk.x, y: py, text: '🧊全屏冰冻 4s!', color: '#29B6F6', timer: 1500, maxTimer: 1500 });
  1037→  }
  1038→  spawnParticles(state, pk.x, py, pk.type === 'sun' ? '#FFD54F' : pk.type === 'double' ? '#FF6F00' : '#29B6F6', 15);
  1039→}
